Ingresa la URL y la API Key Anónima de tu proyecto en Supabase Dashboard (en Project Settings → API).

Script SQL para la tabla 'products' (con moderación) ▼ 
-- Copia y pega esto en el SQL Editor de Supabase:
CREATE TABLE IF NOT EXISTS products (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  title TEXT NOT NULL,
  price NUMERIC NOT NULL,
  category TEXT NOT NULL,
  city TEXT NOT NULL,
  address TEXT,
  lat NUMERIC,
  lng NUMERIC,
  description TEXT,
  seller TEXT NOT NULL,
  phone TEXT NOT NULL,
  image TEXT,
  featured BOOLEAN DEFAULT FALSE,
  estado TEXT DEFAULT 'aprobado'
);

-- Si la tabla ya fue creada anteriormente, agrega las columnas de geolocalización y moderación:
ALTER TABLE products ADD COLUMN IF NOT EXISTS estado TEXT DEFAULT 'aprobado';
ALTER TABLE products ADD COLUMN IF NOT EXISTS address TEXT;
ALTER TABLE products ADD COLUMN IF NOT EXISTS lat NUMERIC;
ALTER TABLE products ADD COLUMN IF NOT EXISTS lng NUMERIC;
UPDATE products SET estado = 'aprobado' WHERE estado IS NULL;

-- Habilitar Row Level Security
ALTER TABLE products ENABLE ROW LEVEL SECURITY;

-- Políticas de lectura e inserción pública
CREATE POLICY "Lectura publica" ON products FOR SELECT USING (true);
CREATE POLICY "Publicacion publica" ON products FOR INSERT WITH CHECK (true);

-- Habilitar Realtime para esta tabla
ALTER PUBLICATION supabase_realtime ADD TABLE products;
